⚙️ PASO 3: Configurar vue-i18n
────────────────────────────────────────────────────────────────────────────

Crea el archivo src/locales/index.js:

import { createI18n } from 'vue-i18n'

// Import translations
import es from './es/menu.json'
import en from './en/menu.json'

// Create i18n instance
const i18n = createI18n({
  legacy: false, // Use Composition API mode
  locale: 'es', // Default language
  fallbackLocale: 'es',
  messages: {
    es: {
      menu: es
    },
    en: {
      menu: en
    }
  }
})

export default i18n

🌐 PASO 6: Cambiar idioma dinámicamente (Opcional)
────────────────────────────────────────────────────────────────────────────

Crea un componente de selector de idioma:

<script setup>
import { useI18n } from 'vue-i18n'

const { locale, availableLocales } = useI18n()

function changeLanguage(lang) {
  locale.value = lang
  // Opcional: guardar en localStorage
  localStorage.setItem('language', lang)
}
</script>

<template>
  <select v-model="locale">
    <option v-for="lang in availableLocales" :key="lang" :value="lang">
      {{ lang.toUpperCase() }}
    </option>
  </select>
</template>
