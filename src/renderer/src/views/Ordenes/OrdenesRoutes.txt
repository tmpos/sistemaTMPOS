import { createRouter, createWebHistory } from 'vue-router'

const routes = [
  // Otras rutas...

  // Rutas de Ordenes
  {
    path: '/ordenes',
    name: 'OrdenesList',
    component: () => import('@/views/ordenes/OrdenesList.vue'),
    meta: {
      title: 'ordenes',
      requiresAuth: true
    }
  },
  {
    path: '/crearOrdenes',
    name: 'OrdenesCreate',
    component: () => import('@/views/ordenes/OrdenesCreate.vue'),
    meta: {
      title: 'Nuevo Ordenes',
      requiresAuth: true
    }
  },
  {
    path: '/editarOrdenes/:id',
    name: 'OrdenesEdit',
    component: () => import('@/views/ordenes/OrdenesEdit.vue'),
    meta: {
      title: 'Editar Ordenes',
      requiresAuth: true
    }
  }
]

const router = createRouter({
  history: createWebHistory(import.meta.env.BASE_URL),
  routes
})

// Guard de navegación (opcional)
router.beforeEach((to, from, next) => {
  // Cambiar título de la página
  if (to.meta.title) {
    document.title = to.meta.title
  }

  // Verificar autenticación

    // if (!isAuthenticated) {
    //   next({ name: 'Login' })
    //   return
    // }
  }

  next()
})

export default router

🔗 NAVEGACIÓN PROGRAMÁTICA
────────────────────────────────────────────────────────────────────────────

Desde cualquier componente, puedes navegar usando:

// Ir a la lista
router.push({ name: 'OrdenesList' })

// Ir a crear
router.push({ name: 'OrdenesCreate' })

// Ir a editar (con ID)
router.push({ name: 'OrdenesEdit', params: { id: '123' } })

// Ir a editar (con objeto completo)
const Ordenes = { id: 123, nombre: 'Juan' }
router.push({ name: 'OrdenesEdit', params: { id: Ordenes.id } })

⚙️ CONFIGURACIÓN ADICIONAL
────────────────────────────────────────────────────────────────────────────

1. Alias de importación:
   Si usas '@/views/...', asegúrate de tener configurado el alias en:
   - vite.config.js (Vite)
   - tsconfig.json (TypeScript)

2. Meta fields:
   - title: Título de la página (para <title>)
   - requiresAuth: Si requiere autenticación
   - breadcrumb: Breadcrumb personalizado
   - permissions: Permisos necesarios

3. Guards de navegación:
   - beforeEach: Se ejecuta antes de cada navegación
   - afterEach: Se ejecuta después de cada navegación
   - beforeEnter: Se ejecuta solo para estas rutas
